ws.onmessage = (msg) => {
  const data = JSON.parse(msg.data);

  if (data.action === "jira.events") {
    switch (data.eventType) {
      case "issue.created":
        console.log("New Issue:", data.issueKey);
        break;

      case "issue.updated":
        console.log("Updated Issue:", data.changes);
        break;

      case "issue.commented":
        console.log("New Comment:", data.comment);
        break;

      default:
        console.log("Unknown Jira event:", data);
    }
  }
};
